```r
# Build a solver_result from iterative solver output.
# All solvers MUST use this function for consistent result structure.
# @param theta Parameter estimate vector
# @param ll Log-likelihood at theta
# @param converged Logical convergence flag
# @param hessian Hessian matrix at theta (or NULL)
# @param solver_name Character name of the solver
# @param iterations Integer iteration count
# @param recorder Trace recorder (from new_trace_recorder) or NULL
# @param n_obs Number of observations (for AIC/BIC) or NULL
# @param diagnostics Named list of solver-specific diagnostics
# @return A solver_result object (inherits mle_fit_numerical, mle_fit)
# @noRd
.build_result <- function(theta, ll, converged, hessian, solver_name,
                          iterations, recorder, n_obs = NULL,
                          diagnostics = list()) {
  sol <- list(
    par = theta,
    value = ll,
    convergence = if (converged) 0L else 1L,
    hessian = hessian
  )

  mle <- algebraic.mle::mle_numerical(sol = sol, superclasses = "solver_result")
  if (!is.null(n_obs)) mle$nobs <- n_obs

  mle$solver_info <- list(
    name = solver_name,
    iterations = as.integer(iterations),
    converged = converged,
    trace_data = finalize_trace(recorder),
    composition = NULL,
    diagnostics = if (length(diagnostics) > 0) diagnostics else NULL
  )

  mle
}
```

```r
#' Fisher Scoring Solver
#'
#' Variant of Newton-Raphson that uses the expected Fisher information
#' instead of the observed Fisher. Can be more stable for some problems.
#'
#' @inheritParams newton_raphson
#' @return A solver function with signature (problem, theta0, trace) ->
#'   mle_result
#'
#' @details
#' Fisher scoring is identical to Newton-Raphson when the expected and
#' observed Fisher information are equal (e.g., exponential families).
#' For other models, it may have different convergence properties.
#'
#' @examples
#' \donttest{
#' set.seed(42)
#' x <- rnorm(50, 5, 2)
#' problem <- mle_problem(
#'   loglike = function(theta) sum(dnorm(x, theta[1], theta[2], log = TRUE)),
#'   constraint = mle_constraint(
#'     support = function(theta) theta[2] > 0,
#'     project = function(theta) c(theta[1], max(theta[2], 1e-8))
#'   )
#' )
#' solver <- fisher_scoring()
#' result <- solver(problem, c(4, 1.5))
#' }
#'
#' @export
fisher_scoring <- function(...) {
  solver <- newton_raphson(...)
  function(problem, theta0, trace = mle_trace()) {
    result <- solver(problem, theta0, trace)
    result$solver_info$name <- "fisher_scoring"
    result
  }
}
```

Replace the current `mle_problem` function definition (lines 55-90) with a generic + default method. The generic dispatches on `loglike`:

```r
#' Create an MLE Problem Specification
#'
#' Encapsulates a maximum likelihood estimation problem, separating the
#' statistical specification from the optimization strategy.
#'
#' When passed a \code{likelihood_model} object (from the \pkg{likelihood.model}
#' package), automatically extracts log-likelihood, score, and Fisher information
#' functions.
#'
#' @param loglike Log-likelihood function taking parameter vector theta, or a
#'   \code{likelihood_model} object.
#' @param ... Additional arguments passed to methods
#' @return An mle_problem object
#'
#' @examples
#' # Direct specification
#' problem <- mle_problem(
#'   loglike = function(theta) -sum((theta - c(1, 2))^2)
#' )
#'
#' @seealso \code{\link{mle_constraint}} for constraint specification
#' @export
mle_problem <- function(loglike, ...) {
  UseMethod("mle_problem")
}

#' @rdname mle_problem
#' @param score Score function (gradient of log-likelihood). If NULL,
#'   computed numerically via numDeriv::grad when needed.
#' @param fisher Fisher information matrix function. If NULL, computed
#'   numerically via numDeriv::hessian when needed.
#' @param constraint Domain constraints as mle_constraint object
#' @param theta_names Character vector of parameter names for nice output
#' @param n_obs Number of observations (for AIC/BIC computation)
#' @param cache_derivatives Logical; if TRUE and score/fisher are computed
#'   numerically, cache the most recent result. Default is FALSE.
#' @export
mle_problem.default <- function(
  loglike,
  score = NULL,
  fisher = NULL,
  constraint = NULL,
  theta_names = NULL,
  n_obs = NULL,
  cache_derivatives = FALSE,
  ...
) {
  stopifnot(is.function(loglike))
  if (!is.null(score)) stopifnot(is.function(score))
  if (!is.null(fisher)) stopifnot(is.function(fisher))
  if (!is.null(constraint)) stopifnot(inherits(constraint, "mle_constraint"))
  if (!is.null(theta_names)) stopifnot(is.character(theta_names))
  if (!is.null(n_obs)) stopifnot(is.numeric(n_obs), n_obs > 0)
  stopifnot(is.logical(cache_derivatives), length(cache_derivatives) == 1)

  if (is.null(constraint)) {
    constraint <- mle_constraint()
  }

  structure(
    list(
      loglike = loglike,
      .score = score,
      .fisher = fisher,
      constraint = constraint,
      theta_names = theta_names,
      n_obs = n_obs,
      cache_derivatives = cache_derivatives,
      .cache = new.env(parent = emptyenv())
    ),
    class = "mle_problem"
  )
}

#' @rdname mle_problem
#' @param data Data frame (or matrix/vector) of observations
#' @export
mle_problem.likelihood_model <- function(
  loglike,
  data,
  constraint = NULL,
  theta_names = NULL,
  cache_derivatives = FALSE,
  ...
) {
  if (!requireNamespace("likelihood.model", quietly = TRUE)) {
    stop("Package 'likelihood.model' is required for this method. ",
         "Install it with: install.packages('likelihood.model')",
         call. = FALSE)
  }
  model <- loglike
  ll_fn <- likelihood.model::loglik(model)
  sc_fn <- likelihood.model::score(model)
  hl_fn <- likelihood.model::hess_loglik(model)

  n_obs <- if (is.data.frame(data) || is.matrix(data)) nrow(data) else length(data)

  # hess_loglik returns Hessian of log-likelihood (negative semi-definite at MLE).
  # mle_problem$fisher expects Fisher info (positive definite) = -Hessian.
  mle_problem.default(
    loglike = function(theta) ll_fn(data, theta),
    score = function(theta) sc_fn(data, theta),
    fisher = function(theta) -hl_fn(data, theta),
    constraint = constraint,
    theta_names = theta_names,
    n_obs = n_obs,
    cache_derivatives = cache_derivatives
  )
}
```

```markdown
# compositional.mle 2.0.0

## Breaking Changes

* All solvers now return `solver_result` objects (class vector:
  `c("solver_result", "mle_fit_numerical", "mle_fit")`). Per-solver subclasses
  (`mle_gradient_ascent`, `mle_bfgs`, etc.) are removed.
* Solver metadata moved to `$solver_info` nested list. Access solver name via
  `result$solver_info$name`, iterations via `result$solver_info$iterations`,
  trace data via `result$solver_info$trace_data`.
* Composition metadata moved to `$solver_info$composition`. Access strategy via
  `result$solver_info$composition$strategy`, chain results via
  `result$solver_info$composition$chain`, etc.
* Solver-specific diagnostics moved to `$solver_info$diagnostics` (e.g.,
  `$diagnostics$final_temp` for sim_anneal, `$diagnostics$cycles` for
  coordinate_ascent).
* `is_mle_numerical()` replaced by `is_solver_result()`.
* `compose()` removed — use `chain()` instead (superset with `early_stop`).
* `mle_problem()` is now an S3 generic dispatching on its first argument.
* Requires `algebraic.mle >= 2.0.0`.

## New Features

* `mle_problem.likelihood_model()`: Bridge to the `likelihood.model` package.
  Pass a `likelihood_model` object and data to auto-extract loglik/score/fisher.
* `fisher_scoring()` now correctly reports `"fisher_scoring"` as solver name
  (previously reported `"newton_raphson"`).
* `coordinate_ascent()` gains `learning_rate` parameter for non-line-search mode.
* `grid_search()` now errors if the grid would exceed 1 million points.
* `race()` warns when `parallel = TRUE` but `future` is not installed.

## Bug Fixes

* `sim_anneal()` now correctly reports convergence based on temperature cooling
  (previously always reported `converged = TRUE`).
* `coordinate_ascent()` non-line-search mode uses scaled gradient step instead
  of hardcoded 0.1 step.
* `print.mle_trace_data()` no longer accesses vector elements before null-check.
* `print.mle_problem()` correctly detects default (unconstrained) problems.
```
